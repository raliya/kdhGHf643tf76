function Protect-ConfigurationSecrets {
    param(
        [AllowEmptyString()]
        [string]$Content,

        [Parameter(Mandatory = $true)]
        [string]$Extension
    )

    if ([string]::IsNullOrEmpty($Content)) {
        return $Content
    }

    # Исходные файлы C# и XAML не изменяются, чтобы не повредить код.
    if ($Extension -in @('.cs', '.xaml', '.csproj', '.sln', '.props', '.targets', '.resx')) {
        return $Content
    }

    $protected = $Content

    # JSON: "Password": "значение"
    $protected = [regex]::Replace(
        $protected,
        '(?im)("(?:password|passwd|pwd|token|secret|apiKey|api_key|accessToken|refreshToken|privateKey)"\s*:\s*")[^"]*(")',
        '$1REMOVED$2'
    )

    # XML: <Password>значение</Password>
    $protected = [regex]::Replace(
        $protected,
        '(?im)(<(?:Password|Passwd|Token|Secret|ApiKey|AccessToken|RefreshToken)>).*?(</(?:Password|Passwd|Token|Secret|ApiKey|AccessToken|RefreshToken)>)',
        '$1REMOVED$2'
    )

    # Формат KEY=VALUE.
    $protected = [regex]::Replace(
        $protected,
        '(?im)^(\s*(?:PASSWORD|PASSWD|PWD|TOKEN|SECRET|API_KEY|APIKEY|ACCESS_TOKEN|REFRESH_TOKEN)\s*=\s*).*$',
        '$1REMOVED'
    )

    # Строки подключения.
    $protected = [regex]::Replace(
        $protected,
        '(?im)(Password|Pwd)\s*=\s*[^;`r`n]*',
        '$1=REMOVED'
    )

    return $protected
}

function Protect-Secrets {
    param(
        [AllowEmptyString()]
        [string]$Content,

        [Parameter(Mandatory = $true)]
        [string]$Extension
    )

    if ([string]::IsNullOrEmpty($Content)) {
        return $Content
    }

    # Исходный код не изменяем, чтобы отчёт содержал точный код.
    if ($Extension -in @(
        '.cs',
        '.xaml',
        '.csproj',
        '.sln',
        '.props',
        '.targets',
        '.resx'
    )) {
        return $Content
    }

    $result = $Content

    # Скрытие типичных секретов в JSON.
    $jsonPattern = '(?im)("(?:password|passwd|pwd|token|secret|apiKey|api_key|accessToken|refreshToken|privateKey)"\s*:\s*")[^"]*(")'
    $result = [regex]::Replace(
        $result,
        $jsonPattern,
        '$1REMOVED$2'
    )

    # Скрытие типичных секретов в XML.
    $xmlPattern = '(?is)(<(?:Password|Passwd|Token|Secret|ApiKey|AccessToken|RefreshToken)>).*?(</(?:Password|Passwd|Token|Secret|ApiKey|AccessToken|RefreshToken)>)'
    $result = [regex]::Replace(
        $result,
        $xmlPattern,
        '$1REMOVED$2'
    )

    # Скрытие секретов в формате KEY=VALUE.
    $envPattern = '(?im)^(\s*(?:PASSWORD|PASSWD|PWD|TOKEN|SECRET|API_KEY|APIKEY|ACCESS_TOKEN|REFRESH_TOKEN)\s*=\s*).*$'
    $result = [regex]::Replace(
        $result,
        $envPattern,
        '$1REMOVED'
    )

    # Скрытие паролей в строках подключения.
    $connectionPattern = '(?im)((?:Password|Pwd)\s*=\s*)[^;\r\n]*'
    $result = [regex]::Replace(
        $result,
        $connectionPattern,
        '$1REMOVED'
    )

    return $result
}

function Assert-File {
    param([string]$Path)

    if (-not (Test-Path -LiteralPath $Path -PathType Leaf)) {
        throw "Не найден файл: $Path"
    }
}

function Read-Source {
    param([string]$Path)

    $text = [System.IO.File]::ReadAllText(
        $Path,
        [System.Text.Encoding]::UTF8
    )

    return $text.Replace(
        "`r`n",
        "`n"
    ).Replace(
        "`r",
        "`n"
    )
}

function Write-Source {
    param(
        [string]$Path,
        [string]$Text
    )

    [System.IO.File]::WriteAllText(
        $Path,
        $Text.Replace("`n", "`r`n"),
        $Utf8NoBom
    )
}

function Replace-Required {
    param(
        [string]$Text,
        [string]$OldValue,
        [string]$NewValue,
        [string]$Description,
        [switch]$AllowAlreadyFixed
    )

    if ($Text.Contains($OldValue)) {
        Write-Host "Исправление: $Description"
        return $Text.Replace(
            $OldValue,
            $NewValue
        )
    }

    if ($AllowAlreadyFixed -and
        $Text.Contains($NewValue)) {
        Write-Host "Уже исправлено: $Description" `
            -ForegroundColor Yellow

        return $Text
    }

    throw "Не найден блок для исправления: $Description"
}
